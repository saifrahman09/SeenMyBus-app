    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/12.17.1/firebase-app.js";
        import { getAuth, signInWithPopup, GoogleAuthProvider, onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/12.17.1/firebase-auth.js";
        import { getDatabase, ref, onValue, set, update } from "https://www.gstatic.com/firebasejs/12.17.1/firebase-database.js";

        const firebaseConfig = {
            apiKey: "AIzaSyCXejNb5wgmZ6KJ3Q4r4BhBqw9KPn7iX5I",
            authDomain: "seenmybus.firebaseapp.com",
            databaseURL: "https://seenmybus-default-rtdb.asia-southeast1.firebasedatabase.app",
            projectId: "seenmybus",
            storageBucket: "seenmybus.firebasestorage.app",
            messagingSenderId: "352466758419",
            appId: "1:352466758419:web:b86ed30eff7223910688e6",
            measurementId: "G-7RF7CK39M9"
        };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getDatabase(app);
        const provider = new GoogleAuthProvider();

        const AUTHORIZED_ADMIN_EMAIL = "rahmansaif822@gmail.com";

        // Deterministic Hardware Fingerprinting
        function getDeterministicDeviceToken() {
            try {
                const canvas = document.createElement('canvas');
                const ctx = canvas.getContext('2d');
                ctx.textBaseline = 'top';
                ctx.font = "14px 'Arial'";
                ctx.fillStyle = '#f60';
                ctx.fillRect(125, 1, 62, 20);
                ctx.fillStyle = '#069';
                ctx.fillText('SeenMyBus,AJU-2026', 2, 15);
                ctx.fillStyle = 'rgba(102, 204, 0, 0.7)';
                ctx.fillText('SeenMyBus,AJU-2026', 4, 17);
                const canvasHash = canvas.toDataURL().slice(-40);

                const rawHardwareString = [
                    navigator.userAgent,
                    screen.width + 'x' + screen.height + 'x' + screen.colorDepth,
                    Intl.DateTimeFormat().resolvedOptions().timeZone,
                    navigator.hardwareConcurrency || 4,
                    navigator.language || 'en',
                    canvasHash
                ].join('###');

                let hash = 0;
                for (let i = 0; i < rawHardwareString.length; i++) {
                    const char = rawHardwareString.charCodeAt(i);
                    hash = ((hash << 5) - hash) + char;
                    hash |= 0;
                }
                return 'dev_' + Math.abs(hash).toString(36);
            } catch(e) {
                return 'dev_fallback_' + screen.width + 'x' + screen.height;
            }
        }

        const currentDeviceToken = getDeterministicDeviceToken();

        const authScreen = document.getElementById('auth-screen');
        const dashboardContent = document.getElementById('dashboard-content');
        const authErrorMsg = document.getElementById('auth-error-msg');
        const btnLogin = document.getElementById('btn-google-login');
        const btnLogout = document.getElementById('btn-logout');

        // Authentication Gatekeeper & Database State Sync
        onAuthStateChanged(auth, async (user) => {
            if (user && user.email === AUTHORIZED_ADMIN_EMAIL) {
                // Register authorized state directly in Firebase Database for this device token
                await set(ref(db, `adminState/${currentDeviceToken}`), {
                    isAuthorized: true,
                    email: user.email,
                    lastLogin: Date.now()
                });

                authScreen.classList.add('hidden');
                dashboardContent.classList.remove('hidden');
                loadLiveFleet();
            } else if (user) {
                // Deny unauthorized email, wipe any accidental state, and sign out
                await set(ref(db, `adminState/${currentDeviceToken}`), null);
                authErrorMsg.textContent = `Access Denied: ${user.email} is not authorized.`;
                authErrorMsg.classList.remove('hidden');
                signOut(auth);
            } else {
                authScreen.classList.remove('hidden');
                dashboardContent.classList.add('hidden');
            }
        });

        btnLogin.onclick = async () => {
            try {
                authErrorMsg.classList.add('hidden');
                await signInWithPopup(auth, provider);
            } catch (e) {
                authErrorMsg.textContent = e.message;
                authErrorMsg.classList.remove('hidden');
            }
        };

        // Complete Logout: Clears Database State + Signs Out + Redirects to Map
        btnLogout.onclick = async () => {
            try {
                // Wipe authorization status from Firebase Realtime Database
                await set(ref(db, `adminState/${currentDeviceToken}`), null);
            } catch(e) {
                console.error("Error clearing admin state in database:", e);
            }
            await signOut(auth);
            window.location.href = './index.html';
        };

        // 1. Admin Slot Override
        document.getElementById('btn-adm-assign').onclick = async () => {
            const routeRaw = document.getElementById('adm-route').value.split('|');
            const busNo = document.getElementById('adm-busno').value.trim();
            const spotId = document.getElementById('adm-spot').value;

            if (!busNo) return alert("Please specify a bus number.");

            const updates = {};
            updates[`activeBuses/${spotId}`] = {
                busNos: [busNo],
                routeNum: routeRaw[0],
                name: routeRaw[1],
                users: 999,
                verifiedByAdmin: true,
                updatedAt: Date.now(),
                updatedBy: currentDeviceToken
            };

            await update(ref(db), updates);
            alert(`Slot ${spotId} successfully locked to Bus ${busNo} (${routeRaw[1]})!`);
            document.getElementById('adm-busno').value = '';
        };

        // 2. Broadcast Notifications
        document.getElementById('btn-adm-send-notif').onclick = async () => {
            const title = document.getElementById('adm-notif-title').value.trim() || "Campus Bus Alert";
            const message = document.getElementById('adm-notif-msg').value.trim();
            if (!message) return alert("Please type a notification message.");

            const newNotifKey = Date.now().toString();
            await set(ref(db, `broadcastNotifications/${newNotifKey}`), {
                title,
                message,
                createdAt: Date.now()
            });

            alert("Push notification broadcasted to all users!");
            document.getElementById('adm-notif-msg').value = '';
        };

        // 3. Live Fleet Synchronization
        function loadLiveFleet() {
            onValue(ref(db, 'activeBuses'), (snap) => {
                const data = snap.val() || {};
                const listEl = document.getElementById('adm-active-list');
                listEl.innerHTML = '';

                const keys = Object.keys(data);
                if (keys.length === 0) {
                    listEl.innerHTML = `<p style="font-size: 13px; color: var(--text-sub);">No active buses currently parked.</p>`;
                    return;
                }

                keys.forEach(spotId => {
                    const item = data[spotId];
                    const buses = item.busNos ? item.busNos.join(', ') : item.busNo;
                    const div = document.createElement('div');
                    div.className = 'live-slot-item';
                    div.innerHTML = `
                        <div>
                            <strong>${spotId.toUpperCase()} • Bus ${buses}</strong>
                            <div><span>Route ${item.routeNum} (${item.name}) • Priority: ${item.users}</span></div>
                        </div>
                        <button class="btn-del-mini" onclick="window.removeSpot('${spotId}')">Depart Slot</button>
                    `;
                    listEl.appendChild(div);
                });
            });
        }

        window.removeSpot = async (spotId) => {
            if (confirm(`Remove and mark departed: ${spotId}?`)) {
                await set(ref(db, `activeBuses/${spotId}`), null);
            }
        };

        // 4. Emergency Shift Purge
        document.getElementById('btn-adm-purge-slots').onclick = async () => {
            if (confirm("Are you sure you want to clear all campus parking slots for the new shift?")) {
                await set(ref(db, 'activeBuses'), null);
                await set(ref(db, 'unassignedBuses'), null);
                alert("Campus map successfully purged!");
            }
        };
